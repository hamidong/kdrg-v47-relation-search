'use strict';

const path = require('node:path');
const fs = require('node:fs');
const {
  KdrgSearchService,
} = require('./src/kdrg-search-service');
const Ui = require('./renderer/ui-formatters');

const service = new KdrgSearchService(
  path.resolve(
    '..',
    'data',
    'kdrg_v47_search_integrated_v3.json',
  ),
);

const app = fs.readFileSync(
  'renderer/app.js',
  'utf8',
);

if (
  !app.includes('detail?.parent_adrg_detail')
  || !app.includes(
    'inheritedFromParent: sourceDetail !== detail',
  )
) {
  throw new Error(
    'generic renderer parent fallback missing',
  );
}

if (
  app.includes(
    "if (detail?.aadrg === 'F2120'",
  )
) {
  throw new Error('F2120 hardcode detected');
}

function uniqueStrings(values) {
  return Ui.uniqueStrings(
    Array.isArray(values) ? values : [],
  );
}

function safeDetail(type, id) {
  try {
    return service.getDetail(type, id)?.detail || null;
  } catch {
    return null;
  }
}

const adrgIds = [
  ...service.recordMaps.ADRG.keys(),
].sort();

if (adrgIds.length !== 1132) {
  throw new Error(
    `ADRG record-map count ${adrgIds.length}`,
  );
}

/*
 * IMPORTANT:
 * recordMaps.ADRG values are search/index records, not the complete
 * user-condition detail objects. Stage63B R2 read those values directly,
 * which made every source_logical_table_ids lookup empty and produced
 * parent count 0. Always dereference with getDetail('ADRG', id).
 */
const parents = new Map();

for (const id of adrgIds) {
  const detail = safeDetail('ADRG', id);
  if (!detail) {
    throw new Error(`ADRG detail missing ${id}`);
  }

  const coverage = Ui.userConditionCoverage(detail);
  const sourceIds = uniqueStrings(
    detail.source_logical_table_ids ?? [],
  );

  if (
    coverage.status !== 'NO_EXPLICIT_CONDITION'
    || detail.condition_ast
    || coverage.table_count
    || sourceIds.length !== 1
  ) {
    continue;
  }

  const escaped = String(id).replace(
    /[.*+?^${}()|[\]\\]/g,
    '\\$&',
  );
  const localPattern = new RegExp(
    `^LT_${escaped}_\\d{3}$`,
    'i',
  );

  if (!localPattern.test(sourceIds[0])) {
    continue;
  }

  const summary = Ui.tableSummaryMap(detail)
    .get(sourceIds[0]);
  const codeCount = Number(
    summary?.summary?.code_count ?? 0,
  );

  if (codeCount <= 0) {
    continue;
  }

  parents.set(id, {
    tableId: sourceIds[0],
    codeCount,
  });
}

if (parents.size !== 479) {
  throw new Error(
    `parent count ${parents.size}`,
  );
}

let candidateChildren = 0;
let inheritedGapShapes = 0;
let parentSummaryResolvable = 0;
let parentIdentityMatched = 0;

for (
  const id
  of [...service.recordMaps.AADRG.keys()].sort()
) {
  const child = safeDetail('AADRG', id);
  if (!child) {
    throw new Error(`AADRG detail missing ${id}`);
  }

  if (!parents.has(child.adrg)) {
    continue;
  }

  candidateChildren += 1;

  const ownSourceIds = uniqueStrings(
    child.source_logical_table_ids ?? [],
  );
  const parent = child.parent_adrg_detail || null;
  const parentSourceIds = uniqueStrings(
    parent?.source_logical_table_ids ?? [],
  );

  if (
    parent
    && String(parent.adrg || '') === String(child.adrg)
  ) {
    parentIdentityMatched += 1;
  }

  if (
    !child.condition_ast
    && ownSourceIds.length === 0
    && parentSourceIds.length === 1
  ) {
    inheritedGapShapes += 1;

    const summary = Ui.tableSummaryMap(parent)
      .get(parentSourceIds[0]);
    const codeCount = Number(
      summary?.summary?.code_count ?? 0,
    );

    if (codeCount > 0) {
      parentSummaryResolvable += 1;
    }
  }
}

if (candidateChildren !== 521) {
  throw new Error(
    `candidate child count ${candidateChildren}`,
  );
}
if (inheritedGapShapes !== 521) {
  throw new Error(
    `inherited gap shape ${inheritedGapShapes}`,
  );
}
if (parentIdentityMatched !== 521) {
  throw new Error(
    `parent identity matched ${parentIdentityMatched}`,
  );
}
if (parentSummaryResolvable !== 521) {
  throw new Error(
    `parent summary resolvable ${parentSummaryResolvable}`,
  );
}

const f2120 = safeDetail('AADRG', 'F2120');
if (!f2120) {
  throw new Error('F2120 detail missing');
}
if (f2120.adrg !== 'F212') {
  throw new Error(
    `F2120 parent mismatch ${f2120.adrg}`,
  );
}
if (
  uniqueStrings(
    f2120.source_logical_table_ids ?? [],
  ).length !== 0
) {
  throw new Error(
    'F2120 unexpectedly owns direct source table',
  );
}

const f212ParentIds = uniqueStrings(
  f2120.parent_adrg_detail
    ?.source_logical_table_ids ?? [],
);
if (
  JSON.stringify(f212ParentIds)
  !== JSON.stringify(['LT_F212_001'])
) {
  throw new Error(
    `F2120 parent source ${JSON.stringify(
      f212ParentIds,
    )}`,
  );
}

const f212Table = safeDetail(
  'TABLE',
  'LT_F212_001',
);
if (!f212Table) {
  throw new Error('LT_F212_001 detail missing');
}
const f212Codes = new Set(
  (f212Table.code_records ?? [])
    .map((row) => row.entity_id),
);
const expectedF212Codes = [
  'O0205',
  'O0206',
  'O0241',
  'O2223',
];
for (const code of expectedF212Codes) {
  if (!f212Codes.has(code)) {
    throw new Error(
      `F212 source code missing ${code}`,
    );
  }
}

for (const code of expectedF212Codes) {
  const response = service.search(
    code,
    'ALL',
    { limit: 100 },
  );
  const types = new Set(
    response.results.map(
      (row) => row.entity_type,
    ),
  );

  if (
    !types.has('CODE')
    || !types.has('AADRG')
    || types.has('ADRG')
    || types.has('TABLE')
  ) {
    throw new Error(
      `${code} search contract ${[
        ...types,
      ].join(',')}`,
    );
  }
}

const p6510 = safeDetail('AADRG', 'P6510');
if (
  !p6510
  || p6510.adrg !== 'P651'
  || !p6510.condition_ast
) {
  throw new Error(
    'P6510 inherited AST regression',
  );
}

const pretty = Ui.buildPrettyConditionTree(
  p6510.condition_ast,
);
if (!pretty) {
  throw new Error(
    'P6510 pretty condition tree missing',
  );
}

console.log(
  '[PASS] Stage63B full impact probe',
  JSON.stringify({
    adrgCount: adrgIds.length,
    parents: parents.size,
    candidateChildren,
    inheritedGapShapes,
    parentIdentityMatched,
    parentSummaryResolvable,
    f212Codes: expectedF212Codes,
    p651TreeKind: pretty.kind,
  }),
);
